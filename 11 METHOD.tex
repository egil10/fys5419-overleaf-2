\section{Methods}\label{section:methods}


\subsection{Mean-Variance Portfolio Optimisation}


\begin{equation}
    \text{Financial Data} \rightarrow (\mu,\Sigma) \rightarrow \text{QUBO} \rightarrow \text{Ising Hamiltonian} \rightarrow \text{Pauli operators} \rightarrow \text{QAOA}
\end{equation}

\begin{equation}
    \min_x \left( -\mu^\top x + \lambda x^\top \Sigma x \right)
\end{equation}

$-\mu^\top x$ maximises expected return and $\lambda x^\top \Sigma x$ penalises portfolio variance (risk).

\begin{equation}
    \sum_{i=1}^{n} x_i = K
\end{equation}

\begin{equation}
    C(x) = -\mu^\top x + \lambda x^\top \Sigma x + A \left( \sum_{i=1}^{n} x_i - K \right)^2
\end{equation}

where $A>0$ is a penalty coefficient chosen large enough that the minimum of $C(x)$ always satisfies the budget constraint.

\begin{equation}
    r_i(t) = \frac{P_i(t+1)-P_i(t)}{P_i(t)} \approx \ln\frac{P_i(t+1)}{P_i(t)}
\end{equation}

\begin{equation}
    \mu_i = \langle r_i \rangle_t \quad \Sigma_{ij} = \langle r_ir_j \rangle - \langle r_i \rangle \langle r_j \rangle
\end{equation}


\subsection{Quadratic Unconstrained Binary Optimisation (QUBO)}
QUBO formulation


\begin{equation}
    \min_{x\in\{0,1\}^n} C(x) = \sum_i Q_{ii} x_i + \sum_{i<j} Q_{ij} x_i x_j
\end{equation}

where for binary variables $x_i^2=x_i$

\begin{equation}
    C(x) = \sum_i \left[ -\mu_i + \lambda\Sigma_{ii} + A(1-2K) \right] x_i + \sum_{i<j}\left[ 2\lambda\Sigma_{ij}+2A\right]x_ix_j + AK^2
\end{equation}

\begin{equation}
    C(x) = \sum_i \underbrace{\bigl[-\mu_i + \lambda\Sigma_{ii} + A(1-2K)\bigr]}_{Q_{ii}}\, x_i + \sum_{i<j} \underbrace{\bigl[2\lambda\Sigma_{ij} + 2A\bigr]}_{Q_{ij}}\, x_i x_j + \underbrace{AK^2}_{\text{const}},
\end{equation}

\begin{equation}
    \boxed{
    Q_{ii} = -\mu_i + \lambda\Sigma_{ii} + A(1-2K) 
    \quad
    Q_{ij} = 2\lambda\Sigma_{ij} + 2A
    \quad
    (i>j)
    }
\end{equation}

\subsection{Ising Hamiltonian Mapping}

Mapping $x_i \in \{0,1\}$ to spin variables $z_i\in\{-1,+1\}$ via 
\begin{equation}
    x_i = \frac{1-z_i}{2}
\end{equation}

Substituting into $C(x)$ and collecting terms by degree giving and Ising Hamiltonian. 
\begin{equation}
    H(z) = \sum_i h_iz_i - \frac{1}{4} \sum_{j \neq i} Q_{ij} 
    \quad
    J_{ij} = \frac{Q_{ij}}{4}
    \quad
    (i<j)
\end{equation}

and the total additive constant is 
\begin{equation}
    c = \text{offset} + \frac{1}{2} \sum_i Q_{ii} + \frac{1}{4} \sum_{i<j} Q_{ij}
\end{equation}

Such that $H(z)+c=C(x)\quad\forall x$
