\begin{table*}[h]
    \centering
    \caption{QAOA circuit components used throughout the report.}
    \label{tab:qaoa_components}
    \begin{tabular}{@{}llp{8cm}@{}}
        \toprule
        \textbf{Component} & \textbf{Notation} & \textbf{Description} \\
        \midrule
        Initial state       & $|+\rangle^{\otimes n}$        & Uniform superposition prepared by Hadamard gates on all $n$ qubits (transverse-field mixer) or the Dicke state $|D^n_K\rangle$ (XY-mixer). \\
        Cost unitary        & $U_C(\gamma)$                  & Phase rotations proportional to the QUBO objective; encodes problem structure into amplitudes. \\
        Mixer unitary       & $U_B(\beta)$                   & $R_X(2\beta)$ rotations on each qubit (transverse-field) or pairwise $XX+YY$ rotations on a ring (XY-mixer). \\
        Ansatz state        & $|\psi(\boldsymbol{\gamma},\boldsymbol{\beta})\rangle$ & Alternating product of $p$ cost and mixer layers. \\
        Cost function       & $E_p(\boldsymbol{\gamma},\boldsymbol{\beta})$          & Expectation value of $H_C$, minimised classically by COBYLA. \\
        Approximation ratio & $r = (E - C_{\max})/(C_{\min} - C_{\max})$   & Quality metric in $[0, 1]$ relative to brute-force optimum. \\
        \bottomrule
    \end{tabular}
\end{table*}
