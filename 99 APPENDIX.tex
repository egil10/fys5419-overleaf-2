\begin{table*}[h]
    \centering
    \caption{QAOA Circuit Components}
    \label{tab:qaoa_components}
    \begin{tabular}{@{}llp{8cm}@{}}
        \toprule
        \textbf{Component} & \textbf{Notation} & \textbf{Description} \\
        \midrule
        Initial state       & $|+\rangle^{\otimes n}$        & Uniform superposition prepared by Hadamard gates on all $n$ qubits \\
        Cost unitary        & $U_C(\gamma)$                  & Phase rotations proportional to the QUBO objective; encodes problem structure \\
        Mixer unitary       & $U_B(\beta)$                   & $R_X(2\beta)$ rotations on each qubit; drives exploration of solution space \\
        Ansatz state        & $|\psi(\boldsymbol{\gamma},\boldsymbol{\beta})\rangle$ & Alternating product of $p$ cost and mixer layers \\
        Cost function       & $F_p(\boldsymbol{\gamma},\boldsymbol{\beta})$          & Expectation value of the cost operator $C$; minimized classically \\
        Approximation ratio & $r = F_p / F_{\mathrm{opt}}$   & Quality metric relative to exact brute-force optimum \\
        \bottomrule
    \end{tabular}
\end{table*}
